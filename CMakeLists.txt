cmake_minimum_required(VERSION 3.16)
project(TermTalk VERSION 0.9.0 LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "TermTalk requires Windows.")
endif()
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(winsock_options INTERFACE)
target_compile_features(winsock_options INTERFACE c_std_11)
if(MSVC)
    target_compile_options(winsock_options INTERFACE /W4 /utf-8)
else()
    target_compile_options(winsock_options INTERFACE -Wall -Wextra -Wpedantic)
    if(MINGW)
        # C11 snprintf and 64-bit formats, independent of legacy msvcrt printf.
        target_compile_definitions(winsock_options INTERFACE __USE_MINGW_ANSI_STDIO=1)
        # GCC 6 otherwise adds its legacy msvcrt check to the C99 MinGW declaration.
        target_compile_options(winsock_options INTERFACE -fno-builtin-snprintf)
    endif()
endif()
add_library(winsock_network STATIC libs/network.c libs/server_loop.c)
target_include_directories(winsock_network PUBLIC libs)
target_link_libraries(winsock_network PUBLIC winsock_options ws2_32)
add_library(winsock_protocol STATIC libs/protocol.c)
target_include_directories(winsock_protocol PUBLIC libs)
target_link_libraries(winsock_protocol PUBLIC winsock_options)
add_library(winsock_chat_server STATIC libs/chat_server.c)
target_include_directories(winsock_chat_server PUBLIC libs)
target_link_libraries(winsock_chat_server PUBLIC winsock_network winsock_protocol)
add_library(winsock_server_book STATIC libs/server_book.c)
target_include_directories(winsock_server_book PUBLIC libs)
target_link_libraries(winsock_server_book PUBLIC winsock_options)
add_library(winsock_server_probe STATIC libs/server_probe.c)
target_link_libraries(winsock_server_probe PUBLIC winsock_server_book winsock_network winsock_protocol)
add_library(winsock_file_transfer STATIC libs/file_transfer.c)
target_link_libraries(winsock_file_transfer PUBLIC winsock_protocol)
add_library(winsock_file_browser STATIC libs/file_browser.c)
target_include_directories(winsock_file_browser PUBLIC libs)
target_link_libraries(winsock_file_browser PUBLIC winsock_options)
add_library(winsock_ui_model STATIC libs/ui_model.c libs/command_catalog.c)
target_include_directories(winsock_ui_model PUBLIC libs)
target_link_libraries(winsock_ui_model PUBLIC winsock_protocol winsock_server_book)
add_library(winsock_history STATIC libs/history_store.c)
target_link_libraries(winsock_history PUBLIC winsock_ui_model)
add_library(winsock_client_transport STATIC libs/client_transport.c)
target_link_libraries(winsock_client_transport PUBLIC winsock_network winsock_protocol)
add_library(winsock_console STATIC libs/console_ui.c)
target_include_directories(winsock_console PUBLIC libs)
target_link_libraries(winsock_console PUBLIC winsock_network winsock_ui_model)
add_library(winsock_chat_client STATIC libs/chat.c)
target_link_libraries(winsock_chat_client PUBLIC winsock_console winsock_history winsock_client_transport winsock_server_probe winsock_file_transfer winsock_file_browser)
add_executable(client apps/client/main.c)
add_executable(server apps/server/main.c)
target_link_libraries(client PRIVATE winsock_chat_client shell32)
target_link_libraries(server PRIVATE winsock_chat_server)
# Local experiment archives are excluded from application targets.

# API documentation is a developer tool, not a runtime dependency.
option(WINSOCK_BUILD_DOCS "Enable the Doxygen documentation target" OFF)
if(WINSOCK_BUILD_DOCS)
    find_package(Doxygen REQUIRED)
    set(DOXYGEN_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/docs")
    set(DOXYGEN_GENERATE_HTML YES)
    set(DOXYGEN_GENERATE_LATEX NO)
    set(DOXYGEN_OUTPUT_LANGUAGE Russian)
    set(DOXYGEN_INPUT_ENCODING UTF-8)
    set(DOXYGEN_OPTIMIZE_OUTPUT_FOR_C YES)
    set(DOXYGEN_EXTRACT_ALL NO)
    set(DOXYGEN_WARN_IF_UNDOCUMENTED YES)
    set(DOXYGEN_WARN_IF_DOC_ERROR YES)
    # Doxygen may normalize INPUT to Windows short paths; match by file name.
    set(DOXYGEN_USE_MDFILE_AS_MAINPAGE "README.md")
    set(DOXYGEN_EXCLUDE_SYMBOLS
        WINSOCK_NETWORK_H WINSOCK_PROTOCOL_H WINSOCK_CHAT_SERVER_H WINSOCK_SERVER_LOOP_H WINSOCK_CHAT_H WINSOCK_CONSOLE_UI_H WIN32_LEAN_AND_MEAN)
    # Explicit inputs keep scratch programs and generated files out of the API.
    doxygen_add_docs(docs
        libs/network.h libs/server_loop.h libs/protocol.h libs/chat_server.h libs/chat.h libs/console_ui.h
        libs/branding.h libs/ui_model.h libs/history_store.h libs/client_transport.h
        libs/server_book.h libs/server_probe.h libs/file_transfer.h
        libs/command_catalog.h libs/file_browser.h
        README.md docs/ARCHITECTURE.md docs/PROTOCOL.md
        CONTRIBUTING.md SECURITY.md docs/README.md docs/DOXYGEN.md docs/ROADMAP.md docs/UI.md
        docs/BUILD-VERIFICATION.md docs/VERSIONING.md docs/SECURITY-DESIGN.md CHANGELOG.md
        COMMENT "Generate TermTalk API documentation")
endif()
